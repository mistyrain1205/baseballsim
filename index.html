// 1. スケジュールを429試合分生成する function initSeason() { teams = teamNames.map(name => ({ name, wins: 0, losses: 0, draws: 0, batters: Array.from({length: 9}, () => ({ name: generateRandomPlayerName(), stats: { ab: 0, hits: 0, hr: 0 } })), pitchers: Array.from({length: 6}, () => ({ name: generateRandomPlayerName(), stats: { ipOuts: 0, er: 0, so: 0 } })) })); fullSeasonSchedule = []; let basePairs = []; for(let i = 0; i < 6; i++) { for(let j = i + 1; j < 6; j++) basePairs.push([i, j]); } // 1チーム143試合にするには、各ペアとの対戦を約28-29回ずつ行う必要がある // 6チームで総当たりなら、1チームあたり5チームと戦う。143試合 ÷ 5チーム = 28.6試合 for(let k = 0; k < 29; k++) { fullSeasonSchedule.push(...basePairs); } // 435試合分できたので、必要な429試合にトリミング fullSeasonSchedule = fullSeasonSchedule.slice(0, 429); totalGamesPlayed = 0; } // 2. 進行ボタンの制限を143*6/2 = 429に変更 function runGames(count) { let limit = Math.min(totalGamesPlayed + count, 429); while (totalGamesPlayed < limit) { let pair = fullSeasonSchedule[totalGamesPlayed]; simulateMatch(teams[pair[0]], teams[pair[1]]); totalGamesPlayed++; } updateUI(); }
